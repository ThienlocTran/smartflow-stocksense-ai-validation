train_size = int(len(sku) * 0.8)

def smape(actual, predicted):
    mask = ~((actual == 0) & (predicted == 0))
    return 100 * np.mean(
        2 * np.abs(actual[mask] - predicted[mask]) /
        (np.abs(actual[mask]) + np.abs(predicted[mask]))
    )
⚠️ Lưu ý: Split theo thứ tự thời gian (không shuffle) để đảm bảo tính hợp lệ của time-series.

🤖 Bước 3 — Train & Predict từng mô hình

Chạy Cell 5, 6, 7 tương ứng

Prophet Meta / Facebook Prophet

from prophet import Prophet

train = sku[["date","sales"]].rename(columns={"date":"ds","sales":"y"})
model = Prophet(daily_seasonality=False)
model.fit(train[:train_size])

forecast = model.predict(
    model.make_future_dataframe(periods=len(sku) - train_size)
)
score_prophet = smape(
    train["y"][train_size:].values,
    forecast["yhat"].tail(len(sku) - train_size).values
)
print(f"Prophet sMAPE: {score_prophet:.2f}%")

ETS Exponential Smoothing (Holt-Winters)

from statsmodels.tsa.holtwinters import ExponentialSmoothing

train_sales = sku["sales"][:train_size].values
test_sales  = sku["sales"][train_size:].values

try:
    ets_model = ExponentialSmoothing(
        train_sales, trend='add', seasonal='mul',
        seasonal_periods=7, damped_trend=True
    ).fit(optimized=True)
except Exception:
    # fallback nếu seasonal='mul' thất bại
    ets_model = ExponentialSmoothing(
        train_sales, trend='add', seasonal='add',
        seasonal_periods=7, damped_trend=True
    ).fit(optimized=True)

score_ets = smape(test_sales, ets_model.forecast(len(test_sales)))
print(f"ETS sMAPE: {score_ets:.2f}%")
⚠️ ETS dùng seasonality 7 ngày (tuần). Nếu data ít hơn 2 chu kỳ (14 ngày) sẽ báo lỗi.

XGBoost Gradient Boosting với lag features

from xgboost import XGBRegressor

sku_xgb = sku.copy()
sku_xgb["dayofweek"] = sku_xgb["date"].dt.dayofweek
sku_xgb["month"]     = sku_xgb["date"].dt.month
for lag in [1, 7, 14]:
    sku_xgb[f"lag_{lag}"] = sku_xgb["sales"].shift(lag)
sku_xgb = sku_xgb.dropna()  # bỏ dòng NaN do shift

features = ["lag_1", "lag_7", "lag_14", "dayofweek", "month"]
X = sku_xgb[features].values
y = sku_xgb["sales"].values
n = int(len(X) * 0.8)

xgb_model = XGBRegressor(n_estimators=100, random_state=42)
xgb_model.fit(X[:n], y[:n])
score_xgb = smape(y[n:], xgb_model.predict(X[n:]))
print(f"XGBoost sMAPE: {score_xgb:.2f}%")
⚠️ Dùng lag 14 ngày → cần tối thiểu ~20 ngày data sau khi dropna.
